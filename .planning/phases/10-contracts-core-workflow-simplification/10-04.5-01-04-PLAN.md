# Code-level guard: ingest code should not reference UI-local file paths.

  | tee /tmp/ingest.json \
  | python -c "import json,sys; d=json.load(sys.stdin).get('document') or {}; print('ok' if d.get('id') else 'bad')" \
  | grep -q ok

# Response contract guard: ingest responses should not expose/require local_path.
python - <<'PY'
import pathlib
